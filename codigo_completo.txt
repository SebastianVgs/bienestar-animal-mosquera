import 'dart:convert';
 
import 'package:flutter/foundation.dart';
import 'package:flutter_dotenv/flutter_dotenv.dart';
import 'package:http/http.dart' as http;
 
class OpenAIService {
  static const String _apiUrl = 'https://api.openai.com/v1/chat/completions';
 
  /// Analiza un reporte (texto + imagen) y devuelve un Map con:
  /// - 'gravedad': "Baja" | "Media" | "Alta" | "Desconocida"
  /// - 'resumen': resumen técnico corto
  static Future<Map<String, dynamic>> analizarReporte(
    String descripcion,
    String imagenUrl,
  ) async {
    final apiKey = dotenv.env['OPENAI_API_KEY'];
 
    if (apiKey == null || apiKey.isEmpty) {
      throw Exception('API Key de OpenAI no encontrada');
    }
 
    final headers = {
      'Content-Type': 'application/json',
      'Authorization': 'Bearer $apiKey',
    };
 
    // El Prompt: Instrucciones estrictas para la IA (Ahora condicionado contra spam)
    final prompt = '''
    Eres un analista experto en bienestar animal trabajando para la Alcaldía de Mosquera.
    Tu tarea es analizar el reporte de un ciudadano y la fotografía adjunta.
    
    REGLA FUNDAMENTAL: Si la descripción describe un caso claro de maltrato o abandono animal, pero la imagen no muestra explícitamente a un animal (puede ser una casa, un patio, un vehículo, o una foto de baja calidad), asume que la descripción es verídica y clasifica la gravedad basándote PRINCIPALMENTE en el texto.
    
    REGLA DE RECHAZO: SOLO debes rechazar el reporte si TANTO la descripción como la imagen son evidentes engaños, spam, o problemas no relacionados (ej: "Semáforo dañado", "Bache en la vía", etc.).
    
    Debes devolver ÚNICAMENTE un objeto JSON válido con estas dos claves:
    1. "gravedad": Clasifica el caso en "Baja", "Media", "Alta" (Alta es riesgo de muerte o abuso extremo). Si el reporte es spam evidente, pon "Rechazado".
    2. "resumen": Un resumen técnico, objetivo y corto (máximo 20 palabras) para las autoridades. Si fue rechazado, escribe "Reporte no pertinente a bienestar animal".
    
    Descripción del ciudadano: $descripcion
    ''';
 
    final body = jsonEncode({
      'model': 'gpt-4o-mini',
      // Obligamos a que responda en JSON puro
      'response_format': {'type': 'json_object'},
      'messages': [
        {
          'role': 'user',
          'content': [
            {'type': 'text', 'text': prompt},
            {
              'type': 'image_url',
              'image_url': {
                'url': imagenUrl,
                // 'low' ahorra tokens y dinero, suficiente para clasificar
                'detail': 'low',
              },
            },
          ],
        },
      ],
      'max_tokens': 150,
    });
 
    try {
      final response = await http
          .post(Uri.parse(_apiUrl), headers: headers, body: body)
          .timeout(const Duration(seconds: 30));
 
      if (response.statusCode == 200) {
        final jsonResponse = jsonDecode(utf8.decode(response.bodyBytes));
        final String iaContent =
            jsonResponse['choices'][0]['message']['content'] as String;
 
        final decoded = jsonDecode(iaContent);
        return Map<String, dynamic>.from(decoded as Map);
      } else {
        debugPrint('Error OpenAI (${response.statusCode}): ${response.body}');
        return {
          'gravedad': 'Desconocida',
          'resumen': 'Error al analizar con IA',
        };
      }
    } catch (e) {
      debugPrint('Error de conexión con OpenAI: $e');
      return {
        'gravedad': 'Desconocida',
        'resumen': 'Error de conexión',
      };
    }
  }
}

import '../models/adopcion_model.dart';

import 'dart:io'; // Importante para la decodificación pesada si fuera necesaria

class AdopcionRepository {
  static final _supabase = Supabase.instance.client;

  // Obtiene la lista desde la tabla animales_adopcion
  static Future<List<AdopcionModel>> getMascotasDisponibles() async {
    try {
      // 1. Obtenemos la respuesta de la red
      final response = await _supabase
          .from('animales_adopcion')
          .select()
          .eq('estado', 'Disponible');

      // 2. Le damos un micro-respiro al hilo principal para que dibuje el ProgressIndicator
      await Future.delayed(const Duration(milliseconds: 100));

      // 3. Verificamos que la respuesta no sea nula y sea una lista
      if (response == null) return [];

      final List<dynamic> dataList = response as List<dynamic>;

      // 4. Mapeamos la lista de forma segura
      final List<AdopcionModel> mascotas = dataList.map((e) {
         try {
           return AdopcionModel.fromMap(e as Map<String, dynamic>);
         } catch(error){
           print("Error mapeando un registro de adopcion: $error");
           // Si un registro falla, no rompemos toda la lista, retornamos un modelo vacío (o lo filtramos luego)
           // Por ahora, para evitar fallos, devolvemos null y filtramos.
           return null;
         }
      })
      .where((element) => element != null) // Quitamos los que fallaron
      .cast<AdopcionModel>()
      .toList();

      return mascotas;
    } catch (e) {
      print('⚠️ Error al obtener mascotas: $e');
      return []; // Si falla, devuelve una lista vacía para no romper la app
    }
  }

  // Función actualizada para subir la foto
  static Future<void> enviarSolicitudAdopcion(
      String idAnimal, 
      String nombreAnimal, 
      String telefono, 
      String correo, 
      String direccion,
      List<String> localImagePaths
  ) async {
    try {
      List<String> imageUrls = [];
      
      // 1. Subir cada imagen al Storage
      if (localImagePaths.isNotEmpty) {
        for (String path in localImagePaths) {
          final file = File(path);
          final fileName = 'vivienda_${DateTime.now().millisecondsSinceEpoch}_${path.split('/').last}';
          
          await _supabase.storage.from('imagenes_adopcion').upload(fileName, file);
          final url = _supabase.storage.from('imagenes_adopcion').getPublicUrl(fileName);
          imageUrls.add(url);
        }
      }

      // 2. Insertar en la tabla con la lista de URLs generada
      await _supabase.from('solicitudes_adopcion').insert({
        'id_animal': idAnimal,
        'nombre_animal': nombreAnimal,
        'telefono_contacto': telefono,
        'correo_electronico': correo,
        'direccion_residencia': direccion,
        'fotos_vivienda_urls': imageUrls, // <--- NUEVA COLUMNA TIPO ARRAY
      });
      print('✅ Solicitud de adopción guardada con éxito');
    } catch (e) {
      print('⚠️ Error al guardar solicitud: $e');
      throw Exception('Error al conectar con BD o Storage');
    }
  }
}

import 'package:supabase_flutter/supabase_flutter.dart';
import '../models/campana_model.dart';

class EsterilizacionRepository {
  static final _supabase = Supabase.instance.client;

  static Future<List<CampanaModel>> getCampanas(String tipo) async {
    try {
      final response = await _supabase
          .from('campanas_bienestar')
          .select()
          .eq('tipo', tipo) // Filtra por 'esterilizacion' o 'desparasitacion'
          .order('created_at', ascending: false);

      return (response as List)
          .map((campana) => CampanaModel.fromJson(campana))
          .toList();
    } catch (e) {
      print('Error al obtener las campañas: $e');
      return [];
    }
  }
}

import 'package:supabase_flutter/supabase_flutter.dart';

class VoluntariadoRepository {
  static final _supabase = Supabase.instance.client;

  static Future<void> enviarPostulacion(VoluntariadoModel postulacion) async {
    try {
      await _supabase.from('hogares_paso').insert(postulacion.toMap());
      print('✅ Postulación a hogar de paso enviada exitosamente');
    } catch (e) {
      print('⚠️ Error al guardar la postulación: $e');
      throw Exception('Fallo al conectar con la base de datos');
    }
  }
}

    if (_formKey.currentState!.validate()) {
      setState(() => _isLoading = true);

      try {

          tipoMascotas: _tieneMascotas ? _tipoMascotasController.text.trim() : null,
        );

        await VoluntariadoRepository.enviarPostulacion(postulacion);

        if (mounted) {
          _mostrarDialogoExito();

  // Consulta todas las imagenes de los carruseles
  static Future<List<String>> getImagenesCarrusel() async {
    try {
      final response = await Supabase.instance.client
          .from('imagenes_carrusel') // Tu nombre real de la tabla
          .select('imagen_url')
          .order('orden', ascending: true); // Ordena según el número que le pongas

      return (response as List).map((e) => e['imagen_url'] as String).toList();
    } catch (e) {
      print('Error cargando carrusel: $e');
      return []; 
    }
  }

  // Consulta todos los eventos, ordenados por fecha de creación
  static Future<List<EventoModel>> getEventos(bool esPasado) async {
    try {
      final response = await _supabase
          .from('educacion_sensibilizacion')
          .select()
          .eq('es_pasado', esPasado)
          .order('creado_en', ascending: false);

      return (response as List<dynamic>)
          .map((json) => EventoModel.fromMap(json))
          .toList();
    } catch (e) {
      print('⚠️ Error al cargar eventos: $e');
      return [];
    }
  }

        telefonoContacto: json['telefono_contacto']?.toString(),

    required String description,

    required String telefono,
    required double? latitude,
    required double? longitude,
    required List<String> localEvidencePaths,
  }) async {
    try {
      print('--- INICIANDO PROCESO DE REPORTE ---');
      List<String> imageUrls = [];
      String primeraImagenUrl = '';

      // 1. Subir foto
      for (String path in localEvidencePaths) {
        final file = File(path);
        final fileName = '${DateTime.now().millisecondsSinceEpoch}_${path.split('/').last}';
        await _supabase.storage.from('imagenes_reportes').upload(fileName, file);
        final imageUrl = _supabase.storage.from('imagenes_reportes').getPublicUrl(fileName);
        imageUrls.add(imageUrl);
      }

      if (imageUrls.isNotEmpty) {
        primeraImagenUrl = imageUrls.first;
      }

      // 2. Llamada a OpenAI
      Map<String, dynamic> analisisIA = {'gravedad': 'Desconocida', 'resumen': 'Sin análisis'};
      if (description.isNotEmpty && primeraImagenUrl.isNotEmpty) {
        analisisIA = await OpenAIService.analizarReporte(description, primeraImagenUrl);
      }

      // 3. Insertar en Base de Datos
      final datosAInsertar = {
        'descripcion': description,
        'direccion': address,
        'telefono_contacto': telefono, // NUEVO DATO A INSERTAR
        'latitud': latitude ?? 4.7093,
        'longitud': longitude ?? -74.2278,
        'evidencia_url': primeraImagenUrl,
        'estado': 'RECIBIDO',
        'nivel_urgencia': analisisIA['gravedad'],
        'ia_gravedad': analisisIA['gravedad'],

class AuthRepository {
  static final _supabase = Supabase.instance.client;

  // Iniciar sesión
  static Future<AuthResponse> signIn(String email, String password) async {
    try {
      final response = await _supabase.auth.signInWithPassword(
        email: email,
        password: password,
      );
      return response;
    } catch (e) {
      throw Exception('Correo o contraseña incorrectos.');
    }
  }

  // Cerrar sesión
  static Future<void> signOut() async {
    await _supabase.auth.signOut();
  }

  // Verificar si hay sesión activa
  static bool get isUserLoggedIn {
    return _supabase.auth.currentUser != null;
  }
}

class AdminReportesRepository {
  static final _supabase = Supabase.instance.client;

  // Obtener reportes en tiempo real
  static Stream<List<Map<String, dynamic>>> obtenerReportesStream() {
    return _supabase
        .from('reportes_maltrato')
        .stream(primaryKey: ['id'])
        .order('creado_en', ascending: false);
  }

  // Actualizar el estado de un reporte
  static Future<void> actualizarEstado(String id, String nuevoEstado) async {
    await _supabase
        .from('reportes_maltrato')
        .update({'estado': nuevoEstado})
        .eq('id', id);
  }

class CensoRepository {

  static Future<void> registrarMascota(CensoModel censo) async {
    try {
      await _supabase.from('censo_mascotas').insert(censo.toMap());
    } catch (e) {
      throw Exception('⚠️ Fallo al guardar en la base de datos: $e');
    }
  }

  static Future<List<CensoModel>> obtenerMascotasCensadas() async {
    try {
      final response = await _supabase
          .from('censo_mascotas')
          .select()
          .order('fecha_registro', ascending: false);

      return (response as List<dynamic>)
          .map((json) => CensoModel.fromJson(json))
          .toList();
    } catch (e) {
      return [];
    }
  }

    try {

        'raza': censo.raza,

        'documento_propietario': censo.documentoPropietario,
        'telefono': censo.telefono,
        'direccion': censo.direccion,
      }).eq('id', censo.id);

  Future<void> _submitCenso() async {
    if (!_formKey.currentState!.validate()) return;

    setState(() => _isSubmitting = true);

    try {
      final censo = CensoModel(
        id: DateTime.now().millisecondsSinceEpoch.toString(),
        especie: _especie,
        nombreMascota: _nombreMascotaController.text.trim(),
        raza: _razaController.text.trim(),
        edad: int.parse(_edadController.text.trim()),
        unidadEdad: _unidadEdad,
        sexo: _sexo,
        nombrePropietario: _nombrePropietarioController.text.trim(),
        documentoPropietario: _documentoController.text.trim(),
        telefono: _telefonoController.text.trim(),
        direccion: _direccionController.text.trim(),
        fechaRegistro: DateTime.now(),
      );

      await CensoRepository.registrarMascota(censo);

      if (!mounted) return;

      // Limpieza de campos
      _nombreMascotaController.clear();
      _razaController.clear();
      _edadController.clear();
      _nombrePropietarioController.clear();
      _documentoController.clear();
      _telefonoController.clear();
      _direccionController.clear();

      setState(() {
        _especie = 'Perro';
        _sexo = 'Macho';
        _unidadEdad = 'Años';
        _selectedTabIndex = 1; // Cambiar automáticamente a la pestaña Mis Mascotas
        _loadMisMascotas();
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Mascota registrada en el censo municipal con éxito.'),
          backgroundColor: AppColors.secondary,
          behavior: SnackBarBehavior.floating,
        ),
      );
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Error al procesar el registro.'),
          backgroundColor: Colors.redAccent,
          behavior: SnackBarBehavior.floating,
        ),
      );
    } finally {
      if (mounted) setState(() => _isSubmitting = false);
    }
  }
